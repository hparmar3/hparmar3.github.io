        \item $D_1$ and $D_2$ are independent
        \item we will measure the distance $= c D_1 + (1-c)D_2 = d$
        \item find constant $c$ such that $0 \leq c \leq 1$ and $Var(cD_1 + (1-c)D_2)$ is minimized
        \item $Var(D_1) = \sigma_1^2$ and $Var(D_2) = \sigma_2^2$ are both finite positive
    \end{itemize}

    \hfill

    $D_i = d + \epsilon_i$

    $D_i = cD_1 + (1-c)D_2 + \epsilon_i$

    $Var(D_i) = Var(cD_1 + (1-c)D_2 + \epsilon_i)$

    $= Var(cD_1 + (1-c)D_2)$

    $= Var(cD_1) + Var((1-c)D_2) + 2Cov(cD_1, (1-c)D_2)$

    $= c^2 Var(D_1) + (1-c)^2 Var(D_2) + 2c(1-c)Cov(D_1, D_2)$

    $= c^2 Var(D_1) + (1-c)^2 Var(D_2) + (2c-2c^2) [Cov(D_1, D_2) = 0]$ since independent 

    $= c^2 \sigma_1^2 + (1-2c+c^2) \sigma_2^2$

    \hfill

    $\frac{d}{dc}(c^2 \sigma_1^2 + (1-2c+c^2) \sigma_2^2) = 0$

    $\frac{d}{dc}(c^2 \sigma_1^2 + \sigma_2^2 - 2c \sigma_2^2 + c^2 \sigma_2^2) = 0$

    $2c \sigma_1^2 - 2 \sigma_2^2 + 2c \sigma_2^2 = 0$

    $2c \sigma_1^2 + 2c \sigma_2^2 = 2 \sigma_2^2$

    $c(\sigma_1^2 + \sigma_2^2) = \sigma_2^2$

    $c = \frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2}$

    \hfill

    $= Var(\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2}D_1, (1-\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2})D_2)$

    $= Var(\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2} D_1) + Var((1- \frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2})D_2) + 2 Cov(\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2}D_1, (1 - \frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2}) D_2)$

    $= (\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2})^2 Var(D_1) + (1 - \frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2})^2 Var(D_2) + 2 (\frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2})(1 - \frac{\sigma_2^2}{\sigma_1^2 + \sigma_2^2}) Cov(D_1, D_2)$

    \begin{exercise}{11}
    \hfill

    \begin{itemize}
        \item independent sequence $X_1, X_2, X_3, ... - N(\mu, \sigma^2)$
        \item for each $n$, the random variable $\overline{X_n} = \frac{1}{n} \sum_{x=1}^n X_k$ denotes the sample means of the first $n$ random variables in the sequence
    \end{itemize}

    \hfill

    \textbf{(a)} Because of the linearity property of covariance
    
    $Cov(X_1 + X_2, X_1 - X_2) = Cov(X_1, X_1) - Cov(X_1, X_2) + Cov(X_2, X_1) - Cov(X_2, X_2)$

    This has the following properties:
    \begin{itemize}
        \item $Cov(X_i, X_i) = Var(X_i) = \sigma^2$
        \item $Cov(X_1, X_2) = Cov(X_2, X_1) = 0$, because $X_1$ and $X_2$ are independent
    \end{itemize}

    \hfill

    $Cov(X_1 + X_2, X_1 - X_2) = \sigma^2 - 0 + 0 - \sigma^2 = 0$

    \hfill

    \textbf{(b)}
    \begin{itemize}
        \item if $1 \leq m < n$ compute $Cov(\overline{X_m}, \overline{X_n})$
    \end{itemize}

    $= cov(\overline{X_m}, \overline{X_n})$

    $= Cov(\frac{1}{m} \sum_{j=1}^{m} X_j, \frac{1}{n} \sum_{k=1}^{m} + \frac{1}{n}\sum_{k=m+1}^{n} X_k)$

    $= \frac{1}{mn} Cov(\sum_{j=1}^{m} X_j, \sum_{k=1}^{m} X_k + \sum_{k=m+1}^{n} X_k)$

    $= \frac{1}{mn} Cov(\sum_{j=1}^{m} X_j, \sum_{k=1}^{m} X_k) + \frac{1}{mn} Cov(\sum_{j=1}^{m} X_j, \sum_{k=m+1}^{n} X_k)$

    $= \frac{1}{mn} \sum_{j=1}^{m} \sum_{k=1}^{m} Cov(X_j, X_k) + \frac{1}{mn} \sum_{j=1}^{m} \sum_{k=m+1}^{n} Cov(X_j, X_k)$

    $Cov(X_j, X_k) = \begin{cases}
        \sigma^2, \, if \, j=k \\
        0, \, if \, j \neq k
    \end{cases}$

    $= \frac{1}{mn} \sum_{j=1}^{m} \sum_{k=1}^{m} Cov(X_j, X_k)$

    $= \frac{1}{mn} \sum_{j=1}^{m} \sigma^2$

    $= \frac{\sigma^2}{n}$

    \hfill

    \textbf{(c)} 
    $Cov(X_i - \overline{X_n}, X_j - \overline{X_n}) = Cov(X_i, X_j) - Cov(X_i, X_n) - Cov(X_n, X_j) + Cov(X_n, X_n)$

    \hfill

    Covariance properties:
    \begin{itemize}
        \item $Cov(X_i, X_j) = 0$ since $i \neq j$, therefore $X_i$ and $X_j$ are independent
        \item $Cov(X_i, X_n) = \frac{\sigma^2}{n}$ form part (b)
        \item $Cov(X_n, X_n) = Var(X_n) = \frac{\sigma^2}{n}$
    \end{itemize}

    \hfill

    $Cov(X_i - X_n, X_j - X_n) = 0 - \frac{\sigma^2}{n} - \frac{\sigma^2}{n} + \frac{\sigma^2}{n}$

    $Cov(X_i - X_n, X_j - X_n) = -\frac{\sigma^2}{n}$

    \hfill
    \end{exercise}


    \begin{exercise}{12}
    \hfill

    \begin{itemize}

    \begin{exercise}{16}
    \hfill

    \begin{itemize}
        \item $N$ people enter an elevator at the ground floor
        \item $N - poisson$ distribution with parameter $\lambda$
        \item No one enters above the ground floor
        \item There are $X$ floors above the ground floor, selected uniformly at random
    \end{itemize}

    \hfill

    P(no one selects floor $i$|$N=n$) = $(1 - \frac{1}{k})^n$

    P(no one selects floor $i$) = $\sum_{n=0}^{\infty} \frac{\lambda^n e^{-\lambda}}{n!} \times (1- \frac{1}{k})^n$

    $= e^{-\lambda} \sum_{n=0}^{\infty} \frac{\lambda^n}{n!}(1 - \frac{1}{k})^n$

    Using Taylor Series for $e^x$

    $= e^{-\lambda} e^{\lambda(1- 1/k)}$

    $= e^{-\lambda} e^{\lambda - \lambda \frac{1}{k}}$

    $= e^{-\lambda \frac{1}{k}}$

    \hfill

    P(elevator stops at floor $i$) = $1 - e^{-\lambda \frac{1}{k}}$

    E(number of stops elevator makes) = $\sum_{i=1}^k (1-e^{-\lambda \frac{1}{k}})$

    $= k (1 - e^{- \lambda \frac{1}{k}})$
        
    \end{exercise}
