    \begin{exercise}{1}
    \hfill

    \begin{itemize}
        \item 5 culprits, 2 did it
    \end{itemize}

    \textbf{Send 3 to detention?}
    \begin{itemize}
        \item number of combinations of 3: $\begin{pmatrix}
            5 \\
            3
        \end{pmatrix} = \frac{5!}{3!2!} = \frac{120}{12} = 10$

        \item favorable outcome: 2 guilty, 1 innocent
        
        $\begin{pmatrix}
            3 \\
            1
        \end{pmatrix} = \frac{3!}{2!1!} = \frac{6}{2} = 3$

        $\frac{number of favorable outcomes}{number of outcomes} = \frac{3}{10} = 0.3$
    \end{itemize}

    \textbf{Send 2 to detention?}
    \begin{itemize}
        \item number of combinations of 2: $\begin{pmatrix}
            5 \\
            2
        \end{pmatrix} = \frac{5!}{2!3!} = \frac{120}{12} = 10$

        \item favorable outcome: 2 guilty $->$ only 1 favorable outcome: $\begin{pmatrix}
            2 \\
            2
        \end{pmatrix} = 1$

        $\frac{number of favorable outcomes}{number of outcomes} = \frac{1}{10} = 0.1$
    \end{itemize}

    \textbf{Send 4 to detention?}
    \begin{itemize}
        \item number of combinations of 4: $\begin{pmatrix}
            5 \\
            4
        \end{pmatrix} = \frac{5!}{4!1!} = \frac{120}{24} = 5$

        \item favorable outcomes 2 guilty, 2 innocent

        At least 1 has to be guilty since there is only 1 child left that isn't chosen and 2 trouble makers.

        $\begin{pmatrix}
            3 \\
            1
        \end{pmatrix} = \frac{3!}{1!2!} = \frac{6}{2} = 3$

        $\frac{number of favorable outcomes}{number of outcomes} = \frac{3}{5} = 0.6$
    \end{itemize}
        
    \end{exercise}

    \begin{exercise}{8}
    \hfill

    $2^m = \sum_{j=0}^{m} \begin{pmatrix}
        m \\
        j
    \end{pmatrix}$

    $$j = k - 1$$
    $$k = j + 1$$
    Therfore,
    $$m = n - 1$$
    $$n = m + 1$$

    $2^{n - 1} = \sum_{k = 1}^{n} \begin{pmatrix}
        n - 1 \\
        k - 1
    \end{pmatrix}$

    $n2^{n - 1} = n \sum_{k = 1}^{n} \begin{pmatrix}
        n - 1 \\
        k - 1
    \end{pmatrix}$

    $n2^{n - 1} = \sum_{k = 1}^{n} n \begin{pmatrix}
        n - 1 \\
        k - 1
    \end{pmatrix}$

    $n2^{n - 1} = \sum_{k = 1}^{n} n \times \frac{(n - 1)!}{(k - 1)!(n - k)!}$

    $n2^{n - 1} = \sum_{k = 1}^{n} \frac{n!}{(k - 1)!(n - k)!}$

    $n2^{n - 1} = \sum_{k = 1}^{n} \frac{k}{k} \times \frac{n!}{(k - 1)!(n - k)!}$

    $n2^{n - 1} = \sum_{k = 1}^{n} k \times \frac{n!}{k!(n - k)!}$

    $n2^{n - 1} = \sum_{k = 1}^{n} k \begin{pmatrix}
        n \\
        k
    \end{pmatrix}$
    \end{exercise}

            7 \\
            4
        \end{pmatrix} = \frac{7!}{4!3!} = 35$

    \end{exercise}



    \begin{exercise}{10}

    \begin{itemize}
        \item 12 long strings
        \item entries can be 1, 2, 3, 4
        \item $4^{12}$ = 16,777,216 sample points equally likely
    \end{itemize}

    \textbf{(a)} $\begin{pmatrix}
        12 \\
        3, 3, 3, 3
    \end{pmatrix} = \frac{12!}{(3!)^4} = 369,600$

    $\frac{369,600}{16,777,216} = 0.0220$

    \textbf{(b)} 
    \begin{itemize}
        \item 3 1's
        \item 9 of 2, 3, and  4s

        $3^9 = 19,683$

        \item orders of these combos: $\begin{pmatrix}
            12 \\
            3, 9
        \end{pmatrix} = \frac{12!}{3!9!} = 220$
    \end{itemize}

    Therefore,

    $19,683 \times 220 = 4,330,260$

    probability of exactly 3 1's = $\frac{4,330,260}{16,777,216} = 0.2581$

    \textbf{(c)}
    \begin{itemize}
        \item 3 1's
        \item 3 2's
        \item 6 3 and 4s

        $2^6 = 64$

        \item orders of these combos: $\begin{pmatrix}
            12 \\
            3, 3, 6
        \end{pmatrix} = \frac{12!}{3!3!6!} = 18,480$

        $64 \times 18,480 = 1,182,720$
    \end{itemize}

    probability of exactly 3 1's and 3 2's = $\frac{1,182,720}{16,777,216} = 0.0705$ 
    \end{exercise}
